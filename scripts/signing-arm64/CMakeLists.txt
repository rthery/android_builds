# libconscrypt_openjdk_jni for linux-arm64, from the same tree as the otatools it extends.
# The OpenJDK jni.h matches what CONSCRYPT_OPENJDK expects. Source lists come from the tree's own external/boringssl/android-sources.cmake.
cmake_minimum_required(VERSION 3.16)
project(conscrypt_openjdk_jni LANGUAGES C CXX ASM)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(BORINGSSL_ROOT ${TREE}/external/boringssl/)
include(${BORINGSSL_ROOT}android-sources.cmake)
# Newer trees list all assembly in one set guarded by arch macros; older ones per platform.
if(DEFINED crypto_sources_asm)
  set(asm ${crypto_sources_asm})
else()
  set(asm ${crypto_sources_linux_aarch64})
endif()
add_library(boringssl STATIC ${crypto_sources} ${asm} ${ssl_sources})
target_include_directories(boringssl PUBLIC ${BORINGSSL_ROOT}src/include)
target_compile_definitions(boringssl PRIVATE BORINGSSL_IMPLEMENTATION OPENSSL_SMALL _XOPEN_SOURCE=700)
target_compile_options(boringssl PRIVATE -fvisibility=hidden -fno-common $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions -fno-rtti>)
set(CONSCRYPT ${TREE}/external/conscrypt)
file(GLOB jni_sources ${CONSCRYPT}/common/src/jni/main/cpp/conscrypt/*.cc)
add_library(conscrypt_openjdk_jni SHARED ${jni_sources})
target_include_directories(conscrypt_openjdk_jni PRIVATE ${CONSCRYPT}/common/src/jni/main/include ${CONSCRYPT}/common/src/jni/unbundled/include ${JDK}/include ${JDK}/include/linux)
target_compile_definitions(conscrypt_openjdk_jni PRIVATE CONSCRYPT_OPENJDK)
target_link_libraries(conscrypt_openjdk_jni PRIVATE boringssl -static-libstdc++ -static-libgcc -Wl,--no-undefined)
